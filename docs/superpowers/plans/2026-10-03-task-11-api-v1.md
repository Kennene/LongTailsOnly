**Cel:** Endpointy z ADR 0006 §7, start aplikacji (lifespan: schemat, seed, zegar, polityka) oraz tryb `auto` wpięty w time-travel i zmianę polityki.

**Architektura:** Routery w `app/api/v1/` są cienkie: walidują DTO, wołają serwisy (Zadania 8–10), **commitują transakcję** i mapują błędy. `ServiceError` obsługuje globalny handler (`{"detail": ...}`). `LastAdminError` jest łapany w endpointach decyzji: najpierw commit wpisu `LAST_ADMIN_BLOCKED`, potem 403. Stan procesu (`session_factory`, `clock`, `policy`) siedzi w `app.state`, a ustawia go lifespan; testy ustawiają go ręcznie.

**Stack:** FastAPI, httpx (testy), SQLAlchemy async.

**Spec:** ADR 0003; ADR 0006 §7; ADR 0007 §4–6; plan główny — Zadanie 11.

**Branch:** `feat/task-11-api-v1` · **Zależności:** 2, 4 (B), 6, 7, 8, 9, 10 · **Odblokowuje:** 19; frontend przechodzi z mocków na prawdziwe API

## Ograniczenia globalne

- Ścieżki, body i odpowiedzi dokładnie jak w ADR 0006 §7 (frontend ma już typy z Zadania 12).
- Aktor decyzji administratora: `settings.admin_login`. Aktor time-travel i polityki: admin (`ADMIN`).
- Każdy time-travel i zmiana polityki zapisują audyt (`TIME_TRAVEL`, `POLICY_CHANGED`).
- Tryb `auto`: `enforce_expired` po każdym time-travel i przy przełączeniu na `auto` (ADR 0007 §6).
- Pliki ≤ 300 linii — jeden router na zasób, testy rozbite na kilka plików.

## Review Focus

- **Audyt blokady ostatniego admina przy odpowiedzi 403:** wpis musi przetrwać, mimo że żądanie kończy się błędem → test `test_revoking_last_admin_returns_403_and_is_audited`.

- **Reset zegara po skokach:** `{"reset": true}` wraca do offsetu 0 → test `test_time_travel_advance_and_reset`.

from fastapi import APIRouter

from app.api.deps import ClockDep, PolicyDep, SessionDep, SettingsDep, VCSDep
from app.domain.enums import ActorType, AuditAction, EnforcementMode
from app.schemas.api import ClockView, TimeTravelRequest
from app.services.audit_service import write_audit_event
from app.services.decision_service import enforce_expired

router = APIRouter(prefix="/simulation", tags=["simulation"])


@router.get("/clock", response_model=ClockView)
async def get_clock(clock: ClockDep) -> ClockView:
    return ClockView(now=clock.get_current_time(), offset_days=clock.offset_days)


@router.post("/time-travel", response_model=ClockView)
async def time_travel(
    body: TimeTravelRequest, clock: ClockDep, session: SessionDep, policy: PolicyDep, vcs: VCSDep,
    settings: SettingsDep,
) -> ClockView:
    before = clock.offset_days
    now = clock.reset() if body.reset else clock.advance(body.days)
    await write_audit_event(
        session, now=now, actor_type=ActorType.ADMIN, actor_id=settings.admin_login, action=AuditAction.TIME_TRAVEL,
        target="simulation-clock", details=f"offset_days {before} -> {clock.offset_days}",
    )
    if policy.mode is EnforcementMode.AUTO:
        await enforce_expired(session, vcs, now)
    await session.commit()
    return ClockView(now=now, offset_days=clock.offset_days)
```

from fastapi import APIRouter

from app.api.deps import ClockDep, PolicyDep, SessionDep, SettingsDep, VCSDep
from app.domain.enums import ActorType, AuditAction, EnforcementMode
from app.schemas.api import PolicyUpdate, PolicyView
from app.services.audit_service import write_audit_event
from app.services.decision_service import enforce_expired

router = APIRouter(prefix="/policy", tags=["policy"])


def _view(policy_mode: EnforcementMode, settings: SettingsDep) -> PolicyView:
    return PolicyView(mode=policy_mode, lease_days=settings.lease_days, warning_days=settings.warning_days)


@router.get("", response_model=PolicyView)
async def get_policy(policy: PolicyDep, settings: SettingsDep) -> PolicyView:
    return _view(policy.mode, settings)


@router.put("", response_model=PolicyView)
async def update_policy(
    body: PolicyUpdate, policy: PolicyDep, session: SessionDep, clock: ClockDep, vcs: VCSDep, settings: SettingsDep
) -> PolicyView:
    now = clock.get_current_time()
    previous, policy.mode = policy.mode, body.mode
    await write_audit_event(
        session, now=now, actor_type=ActorType.ADMIN, actor_id=settings.admin_login,
        action=AuditAction.POLICY_CHANGED, target="enforcement-mode", details=f"{previous} -> {body.mode}",
    )
    if body.mode is EnforcementMode.AUTO:
        await enforce_expired(session, vcs, now)
    await session.commit()
    return _view(policy.mode, settings)

from collections.abc import AsyncIterator
from contextlib import asynccontextmanager

from fastapi import FastAPI

from app.api.github_mock.router import router as github_mock_router
from app.api.v1.errors import service_error_handler
from app.api.v1.router import v1_router
from app.core.config import Settings, get_settings
from app.core.policy import Policy
from app.core.time_provider import TimeProvider
from app.db.seed import seed_demo_data
from app.db.session import create_engine, create_session_factory, init_db
from app.services.errors import ServiceError


def create_app(settings: Settings | None = None) -> FastAPI:
    config = settings or get_settings()

    @asynccontextmanager
    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
        engine = create_engine(config.database_url)
        await init_db(engine)
        app.state.session_factory = create_session_factory(engine)
        app.state.clock = TimeProvider()
        app.state.policy = Policy()
        async with app.state.session_factory() as session:
            await seed_demo_data(session, app.state.clock.get_current_time())
        yield
        await engine.dispose()

    app = FastAPI(title="GitHub Access Lease Governor", lifespan=lifespan)
    app.add_exception_handler(ServiceError, service_error_handler)
    app.include_router(github_mock_router)
    app.include_router(v1_router)
    return app


app = create_app()
```

from fastapi import APIRouter, HTTPException

from app.api.deps import ClockDep, SessionDep, SettingsDep, VCSDep
from app.api.v1.errors import forbid_after_commit
from app.domain.enums import ActorType
from app.models import Lease
from app.ports.vcs_provider import LastAdminError
from app.schemas.api import LeaseDecisionRequest, LeaseView
from app.services.decision_service import apply_decision
from app.services.lease_service import get_lease_view, list_lease_views

router = APIRouter(prefix="/leases", tags=["leases"])


@router.get("", response_model=list[LeaseView])
async def list_leases(session: SessionDep, clock: ClockDep, settings: SettingsDep) -> list[LeaseView]:
    return await list_lease_views(session, clock.get_current_time(), settings.warning_days)


@router.post("/{lease_id}/decision", response_model=LeaseView)
async def decide_lease(
    lease_id: int, body: LeaseDecisionRequest, session: SessionDep, vcs: VCSDep, clock: ClockDep, settings: SettingsDep
) -> LeaseView:
    now = clock.get_current_time()
    lease = await session.get(Lease, lease_id)
    if lease is None:
        raise HTTPException(status_code=404, detail=f"Lease {lease_id} not found")
    try:
        await apply_decision(
            session, vcs, lease=lease, action=body.action, now=now, actor_type=ActorType.ADMIN,
            actor_id=settings.admin_login, justification=body.justification, multiplier=body.multiplier,
            days=body.days, until=body.until,
        )
    except LastAdminError as error:
        await forbid_after_commit(session, error)
    await session.commit()
    return await get_lease_view(session, lease_id, now, settings.warning_days)

async def list_users(session: SessionDep) -> list[UserView]:
    counts = dict(
        (await session.execute(
